└─# cat tmp
import hashlib
import binascii

def generate_pbkdf2(password, salt, iterations=50000, key_length=50):
    # 生成PBKDF2密钥
    derived_key = hashlib.pbkdf2_hmac(
        'sha256',                    # 哈希算法
        password.encode('utf-8'),    # 密码
        salt.encode('utf-8'),        # 盐值 - 这里需要是字符串
        iterations,                  # 迭代次数 50000
        key_length                   # 密钥长度 50字节
    )

    # 转换为十六进制字符串
    hex_key = binascii.hexlify(derived_key).decode('utf-8')
    return hex_key

# 使用正确的盐值
password = "88888888"
salt = "1"  # 从数据库获取的真实盐值
